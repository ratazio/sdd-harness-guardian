## Command and result

```powershell
$ErrorActionPreference = 'Stop'
$guide = (Get-FileHash .harness\references\pearson-design.md -Algorithm SHA256).Hash
$logo = (Get-FileHash .harness\assets\brand\pearson-logo-white.png -Algorithm SHA256).Hash
Add-Type -AssemblyName System.Drawing
$img = [System.Drawing.Image]::FromFile((Resolve-Path '.harness\assets\brand\pearson-logo-white.png'))
$dimensions = "{0}x{1}" -f $img.Width,$img.Height
$img.Dispose()

$briefs = rg --files -g stakeholder-brief.html |
  Where-Object { $_ -notmatch '^(testes[\\/]+mock-runs|scripts[\\/]+fixtures)[\\/]' }
$v2 = @($briefs | Where-Object {
  (Get-Content -LiteralPath $_ -Raw) -match 'data-harness-brief-design=["'']v2["'']'
} | ForEach-Object { $_ -replace '\\','/' } | Sort-Object)
$inventory = Get-Content .harness\references\pearson-brief-migration-inventory.md
$inventoryPaths = @($inventory | ForEach-Object {
  if ($_ -match '^\| `([^`]+stakeholder-brief\.html)` \|') { $Matches[1] }
} | Where-Object { $_ } | Sort-Object)
$fontFiles = @(rg --files .harness | Where-Object { $_ -match '\.(woff2?|ttf|otf)$' })
$remoteRefs = @(rg -n 'fonts\.googleapis\.com|fonts\.gstatic\.com|@import\s+url\(https?://|url\(https?://' .harness\templates .harness\references 2>$null)

if ($guide -ne '6EEE3DC5D2A058D99F57C5F2D92696C61B8C3BB24AB9CCFE76C524EC53C909AE') { throw 'guide hash mismatch' }
if ($logo -ne '8EEE1FA799766BF385A307191D38C361677D442457D7CC0F92E5F3FCCC2282F7') { throw 'logo hash mismatch' }
if ($dimensions -ne '175x53') { throw 'logo dimensions mismatch' }
if ([string]::Join('|', $v2) -ne [string]::Join('|', $inventoryPaths)) { throw 'migration inventory corpus mismatch' }
if ($fontFiles.Count -ne 0 -or $remoteRefs.Count -ne 0) { throw 'unapproved local font or remote reference found' }
```
